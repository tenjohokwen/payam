    String keyPrefix,
    String environment,    // "PROD" or "DEV"

) {
    public enum Action { GENERATED, ROTATED, REVOKED, REACTIVATED }
}

/**
 * Domain event for tenant status/profile changes: suspended, reactivated, email changed, webhook URL changed.
 */
public record TenantStatusChangedEvent(
    String tenantName,
    String tenantEmail,    // nullable — listener handles null
    EventType eventType,
    java.time.Instant occurredAt,
    String oldValue,       // present for EMAIL_CHANGED and WEBHOOK_URL_CHANGED
