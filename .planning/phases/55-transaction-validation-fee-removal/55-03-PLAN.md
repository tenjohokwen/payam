    - The reversed-order test is the canonical deadlock case: without ASC ordering, thread A locks t1 first while thread B locks t3 first, then they cross paths (A wants t3, B wants t1) → cycle

    package com.softropic.payam.disbursement.service;

    import org.junit.jupiter.api.Test;

    import java.io.IOException;
    import java.nio.file.Files;
    import java.nio.file.Path;
    import java.nio.file.Paths;
    import java.util.List;
    import java.util.regex.Matcher;
    import java.util.regex.Pattern;
    import java.util.stream.Stream;

    import static org.assertj.core.api.Assertions.assertThat;

    /**
     * FEE-02 structural regression: any Transaction row written for a disbursement
     * payout (i.e. with {@code flow = DISBURSEMENT}) MUST have {@code feeAmount = 0}
     * and {@code feeRuleId = null}.
     *
     * <p>This test scans {@code src/main/java} for builder chains that set
     * {@code .flow(LedgerFlow.DISBURSEMENT)} or {@code .flow(...DISBURSEMENT)} and
     * asserts that the same chain either explicitly sets the fee fields to zero/null
     * OR omits them (default null/null). The current codebase has zero such call
     * sites — the production Transaction.builder() in TransactionService only
     * creates COLLECTION-flow rows, and the disbursement provider ports
     * (OrangeMoneyPort, MtnMoMoPort) write ledger entries via LedgerService instead
     * of Transaction rows. This test pins that structural property: future code that
     * introduces a DISBURSEMENT-flow Transaction.builder() chain MUST also satisfy
     * FEE-02 or this test will fail.
     */
    class Fee02RegressionTest {

        private static final Path SRC_MAIN_JAVA = Paths.get("src/main/java");

        @Test
        void noTransactionBuilderWithDisbursementFlowViolatesFee02() throws IOException {
            // Pattern matches a Transaction.builder() chain spanning multiple lines
            // up to .build() that includes flow(...DISBURSEMENT).
            Pattern builderChain = Pattern.compile(
                "Transaction\\.builder\\(\\)([\\s\\S]{0,2000}?)\\.build\\(\\)",
                Pattern.MULTILINE);

            try (Stream<Path> paths = Files.walk(SRC_MAIN_JAVA)) {
                List<String> violations = paths
                    .filter(p -> p.toString().endsWith(".java"))
                    .flatMap(p -> {
                        try {
                            String content = Files.readString(p);
                            Matcher m = builderChain.matcher(content);
                            return m.results().map(r -> {
                                String chain = r.group(0);
                                if (!chain.contains("DISBURSEMENT")) return null;
                                // Found a Transaction.builder().flow(...DISBURSEMENT)...build() chain.
                                // FEE-02: must zero feeAmount AND null feeRuleId.
                                boolean zerosFee = chain.contains(".feeAmount(BigDecimal.ZERO)")
                                    || chain.contains(".feeAmount(null)")
                                    || !chain.contains(".feeAmount(");  // omitted = default null
                                boolean nullsRule = chain.contains(".feeRuleId(null)")
                                    || !chain.contains(".feeRuleId(");
                                if (!zerosFee || !nullsRule) {
                                    return p + ":\n" + chain;
                                }
                                return null;
                            }).filter(s -> s != null);
                        } catch (IOException e) {
                            throw new RuntimeException(e);
                        }
                    })
                    .toList();

                assertThat(violations)
                    .as("FEE-02: no flow=DISBURSEMENT Transaction.builder() with non-zero fee")
                    .isEmpty();
            }
        }

        @Test
        void disbursementProviderPortsDoNotCreateTransactionRows() throws IOException {
            // FEE-02 structural property: the disbursement initiate path on both
            // provider ports posts ledger entries via LedgerService — it does NOT
            // create Transaction rows. Verify by static check.
            Path orange = SRC_MAIN_JAVA.resolve("com/softropic/payam/orange/service/OrangeMoneyPort.java");
            Path mtn = SRC_MAIN_JAVA.resolve("com/softropic/payam/mtn/service/MtnMoMoPort.java");

            // Read the initiateDisbursement / initiateCashout methods only — find by
            // method-start regex and check the body up to the next method.
            for (Path port : List.of(orange, mtn)) {
                String content = Files.readString(port);
                // Extract initiateDisbursement / initiateCashout method body
                Pattern methodBody = Pattern.compile(
                    "(?:initiateDisbursement|initiateCashout)\\s*\\([^)]*\\)\\s*\\{([\\s\\S]+?)\\n    \\}",
                    Pattern.MULTILINE);
                Matcher m = methodBody.matcher(content);
                while (m.find()) {
                    String body = m.group(1);
                    assertThat(body)
                        .as("FEE-02: " + port + " disburse method must not create Transaction rows")
                        .doesNotContain("Transaction.builder()")
                        .doesNotContain("transactionRepository.save");
                }
            }
        }
    }
