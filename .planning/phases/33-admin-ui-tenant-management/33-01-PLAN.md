```javascript
// --- Tenant Management ---

/** TENT-05: List tenants with optional status filter. params: { status?, page, size } */
listTenants(params = {}) {
  return api.get('/v1/admin/tenants', { params })
},

/** TENT-06: Get full tenant detail including keys. */
getTenantDetail(tenantRef) {
  return api.get(`/v1/admin/tenants/${tenantRef}`)
},

/** WSEC-03: Get plaintext webhook secret. */
getWebhookSecret(tenantRef) {
  return api.get(`/v1/admin/tenants/${tenantRef}/webhook-secret`)
},

/** TENT-10: Update tenant display name. */
updateTenantName(tenantRef, data) {
  return api.patch(`/v1/admin/tenants/${tenantRef}/name`, data)
},

/** TENT-02: Update tenant email. */
updateTenantEmail(tenantRef, data) {
  return api.patch(`/v1/admin/tenants/${tenantRef}/email`, data)
},

/** TENT-03: Update tenant webhook URL. */
updateTenantWebhookUrl(tenantRef, data) {
  return api.patch(`/v1/admin/tenants/${tenantRef}/webhook-url`, data)
},

/** TENT-04: Suspend tenant (revokes all keys). */
suspendTenant(tenantRef) {
  return api.post(`/v1/admin/tenants/${tenantRef}/suspend`)
},

/** TENT-07: Reactivate tenant. Returns ApiKeyDto with rawKey for new PROD key. */
reactivateTenant(tenantRef) {
  return api.post(`/v1/admin/tenants/${tenantRef}/reactivate`)
},

/** TENT-08: Regenerate webhook secret. */
regenerateWebhookSecret(tenantRef) {
  return api.post(`/v1/admin/tenants/${tenantRef}/webhook-secret`)
},

/** Rotate an API key. Returns ApiKeyDto with rawKey. Uses tenantId (Long), not tenantRef. */
rotateKey(tenantId, keyId) {
  return api.post(`/v1/admin/tenants/${tenantId}/keys/${keyId}/rotate`)
},

/** Revoke an API key. Returns 204. Uses tenantId (Long), not tenantRef. */
revokeKey(tenantId, keyId) {
  return api.delete(`/v1/admin/tenants/${tenantId}/keys/${keyId}`)
},

/** Reactivate a revoked key. Returns 204 (no rawKey). Uses tenantId (Long), not tenantRef. */
reactivateKey(tenantId, keyId) {
  return api.post(`/v1/admin/tenants/${tenantId}/keys/${keyId}/reactivate`)
},
```

The component must:
1. Accept props: `rawKey` (String, required), `modelValue` (Boolean, controls visibility via v-model)
2. Emit events: `update:modelValue`, `close`
3. Use `q-dialog` with `:persistent="true"` — cannot be dismissed by clicking outside or pressing Escape (D-10)
4. Display the raw key in a `q-input readonly outlined` with `:input-style="{ fontFamily: 'monospace' }"` (D-10)
5. Include a "Copy" `q-btn flat` with `icon="content_copy"` that calls `navigator.clipboard.writeText(props.rawKey)` and shows `$q.notify({ type: 'positive', message: 'Key copied to clipboard' })`
6. Include a `q-checkbox v-model="copied"` with label "I have copied the key and stored it safely" (D-11)
7. Include a "Done" `q-btn color="primary"` that is `:disable="!copied"` (D-11)
8. On "Done" click: emit `close` event (parent sets rawKey to null), then emit `update:modelValue` with `false`
9. Include explanatory text above the input: "This key will not be shown again. Copy it now and store it in a secure location." (from Copywriting Contract)
10. Title: `text-h6` "API Key Generated"
11. Card min-width: 500px
12. Use `<script setup>` Composition API, no TypeScript (per project convention)
13. Import `useQuasar` from 'quasar' for the notify call on copy

Full component structure:
```vue
<template>
  <q-dialog :model-value="modelValue" persistent @update:model-value="$emit('update:modelValue', $event)">
    <q-card style="min-width: 500px">
      <q-card-section>
        <div class="text-h6">API Key Generated</div>
      </q-card-section>
      <q-card-section>
        <p>This key will not be shown again. Copy it now and store it in a secure location.</p>
        <q-input
          :model-value="rawKey"
          readonly
          outlined
          :input-style="{ fontFamily: 'monospace' }"
        >
          <template #append>
            <q-btn flat icon="content_copy" @click="copyKey" />
          </template>
        </q-input>
      </q-card-section>
      <q-card-section>
        <q-checkbox v-model="copied" label="I have copied the key and stored it safely" />
      </q-card-section>
      <q-card-actions align="right">
        <q-btn color="primary" label="Done" :disable="!copied" @click="dismiss" />
      </q-card-actions>
    </q-card>
  </q-dialog>
</template>

<script setup>
import { ref } from 'vue'
import { useQuasar } from 'quasar'

const $q = useQuasar()
const props = defineProps({
  rawKey: { type: String, required: true },
  modelValue: { type: Boolean, default: false },
})
const emit = defineEmits(['update:modelValue', 'close'])
const copied = ref(false)

function copyKey() {
  navigator.clipboard.writeText(props.rawKey)
  $q.notify({ type: 'positive', message: 'Key copied to clipboard' })
}

function dismiss() {
  emit('close')
  emit('update:modelValue', false)
  copied.value = false
}
</script>
```
