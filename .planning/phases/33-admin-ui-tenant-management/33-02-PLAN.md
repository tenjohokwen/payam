Script setup logic:
```javascript
import { ref, onMounted } from 'vue'
import { useRouter } from 'vue-router'
import { useQuasar } from 'quasar'
import { adminApi } from 'src/api/admin.api'

const $q = useQuasar()
const router = useRouter()
const loading = ref(false)
const rows = ref([])
const statusFilter = ref('ALL')

const pagination = ref({
  page: 1,
  rowsPerPage: 20,
  rowsNumber: 0,
})

// CRITICAL: Spring Page is 0-indexed, Quasar q-table is 1-indexed. Subtract 1.
async function onRequest({ pagination: p }) {
  loading.value = true
  try {
    const params = {
      page: p.page - 1,
      size: p.rowsPerPage,
    }
    if (statusFilter.value !== 'ALL') {
      params.status = statusFilter.value
    }
    const resp = await adminApi.listTenants(params)
    rows.value = resp.data.content
    pagination.value = {
      ...p,
      rowsNumber: resp.data.totalElements,
    }
  } catch {
    $q.notify({ type: 'negative', message: 'Failed to load tenants' })
  } finally {
    loading.value = false
  }
}

// @row-click signature: (evt, row, index) — first arg is native event, not row (Pitfall 5)
function onRowClick(evt, row) {
  router.push(`/admin/tenants/${row.tenantRef}`)
}

onMounted(() => {
  onRequest({ pagination: pagination.value })
})
```
