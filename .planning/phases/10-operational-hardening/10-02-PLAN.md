    STEP 6 — AlertEvaluationService:

    @Service. Inject MeterRegistry meterRegistry, AlertRuleCache alertRuleCache,
    ApplicationEventPublisher eventPublisher. NOT @Transactional (reads in-memory counters only).

    Constants: private static final int MINIMUM_SAMPLE_SIZE = 10;

    @Scheduled(fixedDelayString = "${alert.evaluation-interval-ms:30000}")
    public void evaluate():
      For each rule in alertRuleCache.getCachedRules():
        double actualValue = computeMetricValue(rule.getMetricName());
        if (actualValue < 0) continue; // metric not computable (e.g. no samples yet)
        if (actualValue >= rule.getThreshold().doubleValue()) {
            eventPublisher.publishEvent(new AlertFiredEvent(
                rule.getMetricName(), actualValue,
                rule.getThreshold().doubleValue(), rule.getNotificationChannel()));
        }

    private double computeMetricValue(String metricName) returns:
      "FAILURE_RATE":
        double failed = getCounter("payment.failed.total");
        double success = getCounter("payment.success.total");
        double total = failed + success;
        if (total < MINIMUM_SAMPLE_SIZE) return -1.0; // Pitfall 8 guard
        return failed / total;
      "FRAUD_SPIKE_RATE":
        double fraud = getCounter("payment.fraud.blocked.total");
        double success = getCounter("payment.success.total");
        double total = fraud + success;
        if (total < MINIMUM_SAMPLE_SIZE) return -1.0;
        return fraud / total;
      "CALLBACK_ANOMALY": return -1.0 (not yet implemented — reserved metric name)
      default: return -1.0

    private double getCounter(String name):
      try { return meterRegistry.counter(name).count(); }
      catch (Exception e) { return 0.0; }

    STEP 7 — AlertNotificationListener:

    @Component. Method annotated @EventListener(AlertFiredEvent.class):

    Email subject: "Payam Alert: " + event.metricName() + " threshold breached"
    Email body: String.format("Metric %s actual value %.4f exceeded threshold %.4f",
        event.metricName(), event.actualValue(), event.threshold())
