    ```java
    int THREADS = 20;
    CyclicBarrier barrier = new CyclicBarrier(THREADS);
    ExecutorService pool = Executors.newFixedThreadPool(THREADS);
    AtomicInteger successes = new AtomicInteger(0);
    AtomicInteger insufficients = new AtomicInteger(0);
    AtomicInteger other = new AtomicInteger(0);
    List<Throwable> failures = Collections.synchronizedList(new ArrayList<>());

    for (int i = 0; i < THREADS; i++) {
        pool.submit(() -> {
            RestTemplate rt = noErrorRestTemplate();
            String idem = "RACE-" + UUID.randomUUID();
            try {
                barrier.await(15, TimeUnit.SECONDS);
                ResponseEntity<String> r = postDisbursementWithKey(rt, idem);
                int code = r.getStatusCode().value();
                if (code == 202 && "PROCESSING".equals(parseStatus(r.getBody()))) {
                    successes.incrementAndGet();
                } else if (code == 422 && "INSUFFICIENT_BALANCE".equals(parseErrorCode(r.getBody()))) {
                    insufficients.incrementAndGet();
                } else {
                    other.incrementAndGet();
                }
            } catch (Throwable t) {
                failures.add(t);
            }
        });
    }
    pool.shutdown();
    assertThat(pool.awaitTermination(120, TimeUnit.SECONDS)).as("all threads complete").isTrue();
    assertThat(failures).as("no thread threw").isEmpty();

    // Race outcome
    assertThat(successes.get()).as("exactly 1 PROCESSING").isEqualTo(1);
    assertThat(insufficients.get()).as("19 INSUFFICIENT_BALANCE").isEqualTo(THREADS - 1);
    assertThat(other.get()).as("no other status codes").isZero();

    // BAL-01 wallet invariant — no overdraft
    MerchantWalletBalance wallet = walletRepo.findByTenantId(tenantId).orElseThrow();
    assertThat(wallet.getBalance()).as("balance is 0 after winner").isEqualByComparingTo(BigDecimal.ZERO);
    assertThat(wallet.getReservedAmount()).as("reserved == winner principal").isEqualByComparingTo(PRINCIPAL);

    // Provider called exactly once (only by the winner)
    mtnServer.verify(1, postRequestedFor(urlPathEqualTo("/v1_0/transfer")));

    // Disbursement table has exactly 1 PROCESSING row, 19 FAILED rows
    Integer processingRows = jdbcTemplate.queryForObject(
        "SELECT count(*) FROM main.disbursement WHERE tenant_id = ? AND disbursement_status = 'PROCESSING'",
        Integer.class, tenantId);
    assertThat(processingRows).as("1 PROCESSING row").isEqualTo(1);
    ```

    NOTE on the 19 FAILED rows: depending on the orchestrator's design, INSUFFICIENT_BALANCE may not persist a row. Do NOT assert exact FAILED count; the load-bearing assertion is `processingRows == 1` and the wallet invariant.
